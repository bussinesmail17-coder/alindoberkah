        function processNativeCamera(inputElement, previewId, dataId, watermarkColor) {
            if (!inputElement.files || !inputElement.files[0]) return;
            const file = inputElement.files[0];
            const now = new Date().getTime();
            
            if (now - file.lastModified > 120000) {
                showToast("DITOLAK: Dilarang pakai foto dari Galeri!", true);
                inputElement.value = ""; document.getElementById(previewId).classList.add('hidden'); return;
            }

            const reader = new FileReader();
            reader.onload = function(e) {
                const img = new Image();
                img.onload = function() {
                    const canvas = document.createElement('canvas');
                    let w = img.width, h = img.height;
                    if (w > 800) { h = Math.round((800 / w) * h); w = 800; }
                    canvas.width = w; canvas.height = h;
                    const ctx = canvas.getContext('2d');
                    ctx.drawImage(img, 0, 0, w, h);
                    
                    ctx.fillStyle = "rgba(0, 0, 0, 0.6)"; ctx.fillRect(0, h - 40, w, 40);
                    ctx.font = "bold 14px monospace"; ctx.fillStyle = watermarkColor;
                    ctx.fillText("LIVE: " + new Date().toLocaleString('id-ID'), 15, h - 15);

                    const finalData = canvas.toDataURL('image/jpeg', 0.6);
                    document.getElementById(previewId).src = finalData; document.getElementById(previewId).classList.remove('hidden'); document.getElementById(dataId).value = finalData;
                }; img.src = e.target.result;
            }; reader.readAsDataURL(file);
        }

            const f = fleets.find(x => x.plat === plat); if(!f) return;
            document.getElementById('editOldPlat').value = f.plat; document.getElementById('editFleetPlat').value = f.plat; document.getElementById('editFleetJenis').value = f.jenis; document.getElementById('editFleetDriver').value = f.driver; document.getElementById('editFleetStatus').value = f.status; document.getElementById('editFleetKm').value = f.km || 0; document.getElementById('editFleetBbm').value = f.bbm || 0;
            document.getElementById('editFleetModal').classList.remove('hidden');
        }

        function handleFleetFormSubmit(e) {
            e.preventDefault();
            const btn = document.getElementById('saveFleetBtn'); btn.disabled = true; btn.innerText = "MENYIMPAN...";
            const km = parseFloat(document.getElementById('fleetKm').value) || 0;
            const bbm = parseFloat(document.getElementById('fleetBbm').value) || 0;
            
            // Logic BBM (1 L = 9 KM)
            let indikator = '-';
            if(bbm > 0) { const ratio = km / bbm; indikator = ratio >= 9 ? 'EFISIEN' : 'BOROS'; }

            const data = { plat: document.getElementById('fleetPlat').value, jenis: document.getElementById('fleetJenis').value, driver: document.getElementById('fleetDriver').value || '-', status: document.getElementById('fleetStatus').value, km: km, bbm: bbm, indikator: indikator };
            fleets.unshift(data); renderFleets();

            if (isGASAvailable) {
                google.script.run.withFailureHandler(err => { btn.disabled = false; btn.innerText = "Simpan"; showToast(err.message, true); fleets.shift(); renderFleets(); })
                .withSuccessHandler(res => { btn.disabled = false; btn.innerText = "Simpan"; if(res.success) { showToast(res.message); closeFleetModal(); refreshData(); } else { showToast(res.message, true); fleets.shift(); renderFleets(); } }).addFleet(data, currentUser.role);
            } else { btn.disabled = false; btn.innerText = "Simpan"; closeFleetModal(); showToast("Armada Disimpan!"); }
        }

        function handleEditFleetSubmit(e) {
            e.preventDefault();
            const btn = document.getElementById('updateFleetBtn'); btn.disabled = true; btn.innerText = "UPDATE...";
            const km = parseFloat(document.getElementById('editFleetKm').value) || 0;
            const bbm = parseFloat(document.getElementById('editFleetBbm').value) || 0;
            
            let indikator = '-';
            if(bbm > 0) { const ratio = km / bbm; indikator = ratio >= 9 ? 'EFISIEN' : 'BOROS'; }

            const data = { oldPlat: document.getElementById('editOldPlat').value, plat: document.getElementById('editFleetPlat').value, jenis: document.getElementById('editFleetJenis').value, driver: document.getElementById('editFleetDriver').value || '-', status: document.getElementById('editFleetStatus').value, km: km, bbm: bbm, indikator: indikator };
            const index = fleets.findIndex(x => x.plat === data.oldPlat); if(index !== -1) { fleets[index] = data; renderFleets(); }

            if (isGASAvailable) {
                google.script.run.withFailureHandler(err => { btn.disabled = false; btn.innerText = "Update Data"; showToast(err.message, true); })
                .withSuccessHandler(res => { btn.disabled = false; btn.innerText = "Update Data"; if(res.success) { showToast(res.message); closeEditFleetModal(); refreshData(); } else { showToast(res.message, true); } }).updateFleet(data);
            } else { btn.disabled = false; btn.innerText = "Update Data"; closeEditFleetModal(); }
        }

        function renderFleets() {
            const container = document.getElementById('fleetCardsContainer'); if(!container) return; container.innerHTML = '';
            if(fleets.length === 0) { container.innerHTML = '<p class="text-slate-500 text-xs font-mono col-span-full">Belum ada armada.</p>'; return; }
            fleets.forEach(f => {
                const statusColor = f.status === 'Aktif' ? 'bg-emerald-500/20 text-emerald-400 border-emerald-500/30' : 'bg-rose-500/20 text-rose-400 border-rose-500/30';
                const indiColor = f.indikator === 'EFISIEN' ? 'text-emerald-400 bg-emerald-500/10 border border-emerald-500/30' : (f.indikator === 'BOROS' ? 'text-rose-400 bg-rose-500/10 border border-rose-500/30' : 'text-slate-400 bg-slate-800');
                
                container.innerHTML += `
                <div class="cyber-glass p-5 rounded-2xl border border-amber-500/20 space-y-4 hover:border-amber-400 transition-colors flex flex-col">
                    <div class="flex justify-between items-start">
                        <div><span class="text-[9px] text-amber-400 uppercase tracking-widest font-bold">${f.jenis}</span><h4 class="font-black text-white text-lg font-orbitron mt-0.5">${f.plat}</h4></div>
                        <span class="px-2 py-0.5 rounded text-[9px] font-bold border ${statusColor}">${f.status}</span>
                    </div>
                    <div class="text-xs text-slate-300 font-mono flex items-center gap-2 flex-1"><i class="fa-solid fa-user-astronaut text-slate-500"></i> Driver: <span class="text-amber-300 font-bold">${f.driver || '-'}</span></div>
                    <div class="grid grid-cols-2 gap-2 text-[10px] text-slate-300 font-mono bg-slate-900/80 p-3 rounded-xl border border-slate-800">
                        <div><i class="fa-solid fa-road text-amber-400 mr-1"></i> <span class="font-bold text-white">${f.km || 0} KM</span></div>
                        <div><i class="fa-solid fa-gas-pump text-rose-400 mr-1"></i> <span class="font-bold text-white">${f.bbm || 0} L</span></div>

                </div>`;
            });
        }

        function handleFormSubmit(e) { 
            e.preventDefault(); 
            const btn = document.getElementById('submitTxBtn'); btn.disabled = true; btn.innerText = "MENYIMPAN...";
            const txData = { date: document.getElementById('tanggal').value, type: document.querySelector('input[name="tipe"]:checked').value, category: document.getElementById('kategori').value, amount: Number(document.getElementById('jumlah').value), driver: document.getElementById('driver').value || '-', method: document.getElementById('metode').value, status: document.getElementById('status').value, notes: document.getElementById('catatan').value };
            transactions.unshift({ id: 'TRX-' + Date.now().toString(36).toUpperCase(), ...txData }); renderTransactionsTable(); 
            showToast("Transaksi Otomatis Masuk Ledger!"); document.getElementById('transactionForm').reset(); 
            const dateInput = document.getElementById('tanggal'); if (dateInput) dateInput.value = new Date().toISOString().split('T')[0];

            if (isGASAvailable) {
                google.script.run.withFailureHandler(err => { btn.disabled = false; btn.innerText = "SIMPAN TRANSAKSI INSTAN"; showToast("Koneksi gagal!", true); transactions.shift(); renderTransactionsTable(); })
                .withSuccessHandler(res => { btn.disabled = false; btn.innerText = "SIMPAN TRANSAKSI INSTAN"; if (res.success) { refreshData(); } else { showToast(res.message, true); transactions.shift(); renderTransactionsTable(); } }).addTransaction(txData);
            } else { btn.disabled = false; btn.innerText = "SIMPAN TRANSAKSI INSTAN"; }
        }

        function renderTransactionsTable() {
            const tbody = document.getElementById('transactionTableBody'); if(!tbody) return; tbody.innerHTML = '';
            if(transactions.length === 0) { tbody.innerHTML = '<tr><td colspan="4" class="p-4 text-center text-slate-500 font-bold font-mono">Belum ada transaksi</td></tr>'; }
            let income = 0, expense = 0, pending = 0;
            transactions.forEach(t => {
                const tipeStr = t.type || t.tipe; const jumlahStr = Number(t.amount || t.jumlah); const tglStr = t.date || t.tanggal; const katStr = t.category || t.kategori;
                if(tipeStr === 'Pemasukan') income += jumlahStr; if(tipeStr === 'Pengeluaran') expense += jumlahStr; if(t.status === 'Pending') pending++;
                const color = tipeStr === 'Pemasukan' ? 'text-emerald-400' : 'text-rose-400'; const statusColor = t.status === 'Lunas' ? 'bg-emerald-950 text-emerald-400 border border-emerald-500/30' : 'bg-amber-950 text-amber-400 border border-amber-500/30';
                tbody.innerHTML += `<tr class="hover:bg-slate-900/50 border-b border-slate-800 transition-colors"><td class="p-4 text-slate-300 font-bold">${tglStr}<br><span class="text-[9px] text-slate-500 font-mono">${t.id}</span></td><td class="p-4 text-white">${katStr}<br><span class="text-[10px] ${color} font-black uppercase tracking-wider">${tipeStr}</span></td><td class="p-4 text-right font-black font-orbitron ${color}">Rp ${jumlahStr.toLocaleString('id-ID')}</td><td class="p-4 text-center"><span class="px-2 py-1 rounded-md text-[10px] font-bold ${statusColor}">${t.status}</span></td></tr>`;
            });
            if(document.getElementById('statTotalIncome')) document.getElementById('statTotalIncome').innerText = 'Rp ' + income.toLocaleString('id-ID');
            if(document.getElementById('statTotalExpense')) document.getElementById('statTotalExpense').innerText = 'Rp ' + expense.toLocaleString('id-ID');
            if(document.getElementById('statNetProfit')) document.getElementById('statNetProfit').innerText = 'Rp ' + (income - expense).toLocaleString('id-ID');
            if(document.getElementById('statPending')) document.getElementById('statPending').innerText = pending + ' Trx';
        }
